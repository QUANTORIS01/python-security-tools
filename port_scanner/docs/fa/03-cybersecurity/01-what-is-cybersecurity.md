# جلسه ۱: امنیت سایبری چیست؟

# مقدمه

تا اینجا دو فصل مهم را پشت سر گذاشته‌ایم:

---

## فصل اول

```text
Python
```

یاد گرفتیم چگونه با پایتون برنامه بنویسیم.

---

## فصل دوم

```text
Networking
```

یاد گرفتیم کامپیوترها چگونه با یکدیگر ارتباط برقرار می‌کنند.

---

اکنون وارد فصل سوم شده‌ایم:

```text
Cybersecurity
```

---

در این فصل قرار است یاد بگیریم:

```text
چگونه از سیستم‌ها محافظت کنیم
```

و همچنین:

```text
مهاجمان چگونه فکر می‌کنند
```

---

# امنیت سایبری چیست؟

امنیت سایبری یا:

```text
Cybersecurity
```

دانش محافظت از:

<p dir='rtl'>
• کامپیوترها <br>
• سرورها <br>
• شبکه‌ها <br>
• نرم‌افزارها <br>
• داده‌ها
</p>

در برابر تهدیدها و حملات است.

---

# یک مثال ساده

فرض کن یک خانه داری.

---

برای محافظت از آن از موارد زیر استفاده می‌کنی:

```text
قفل
درب
دوربین
نگهبان
سیستم هشدار
```

---

در دنیای دیجیتال نیز همین اتفاق رخ می‌دهد.

---

به جای خانه داریم:

```text
کامپیوتر
سرور
وب‌سایت
شبکه
```

---

و به جای قفل داریم:

```text
Firewall
Antivirus
Encryption
Authentication
```

---

# چرا امنیت سایبری مهم است؟

امروزه تقریباً همه چیز دیجیتالی شده است.

---

مثال:

```text
بانکداری
پزشکی
دانشگاه
شرکت‌ها
دولت‌ها
فروشگاه‌ها
```

---

اگر امنیت وجود نداشته باشد:

---

ممکن است:

```text
اطلاعات سرقت شوند
```

---

یا:

```text
سرویس‌ها از کار بیفتند
```

---

یا:

```text
داده‌ها نابود شوند
```

---

# هدف اصلی امنیت سایبری

معمولاً امنیت بر سه اصل مهم استوار است.

---

به این سه اصل می‌گویند:

```text
CIA Triad
```

---

<p dir='rtl' style='font-size:32px; font-wight:bold;'>
Confidentiality
</p>

محرمانگی

---

یعنی:

```text
فقط افراد مجاز
```

باید به اطلاعات دسترسی داشته باشند.

---

مثال:

```text
رمز حساب بانکی
```

---

نباید برای همه قابل مشاهده باشد.

---

<p dir='rtl' style='font-size:32px; font-wight:bold;'>
Integrity
</p>

یکپارچگی

---

یعنی:

```text
اطلاعات نباید تغییر کنند
```

---

مثال:

---

اگر موجودی حساب:

```text
1000$
```

باشد،

نباید کسی آن را به:

```text
100000$
```

تغییر دهد.

---

<p dir='rtl' style='font-size:32px; font-wight:bold;'>
Availability
</p>

دردسترس بودن

---

یعنی:

```text
سرویس باید در دسترس باشد
```

---

مثال:

---

وب‌سایت بانک باید زمانی که کاربر نیاز دارد باز شود.

---

اگر سرور از کار بیفتد:

---

<p dir='rtl'>
Availability از بین رفته است.
</p>

---

# امنیت سایبری فقط برای هکرها نیست

بسیاری از افراد تصور می‌کنند:

```text
Cybersecurity = Hacking
```

---

اما این تصور اشتباه است.

---

بخش بزرگی از امنیت شامل:

```text
دفاع
مدیریت
آموزش
تحلیل
پیشگیری
```

است.

---

# امنیت سایبری شامل چه حوزه‌هایی می‌شود؟

<p dir='rtl' style='font-size:24px; font-wight:bold;'>
Network Security
</p>

امنیت شبکه

---

محافظت از:

```text
شبکه‌ها
روترها
سوئیچ‌ها
فایروال‌ها
```

---

<p dir='rtl' style='font-size:24px; font-wight:bold;'>
Application Security
</p>

امنیت نرم‌افزار

---

محافظت از:

```text
وب‌سایت‌ها
برنامه‌ها
APIها
```

---

<p dir='rtl' style='font-size:24px; font-wight:bold;'>
Cloud Security
</p>

امنیت فضای ابری

---

محافظت از:

```text
AWS
Azure
Google Cloud
```

---

<p dir='rtl' style='font-size:24px; font-wight:bold;'>
Endpoint Security
</p>

امنیت دستگاه‌ها

---

محافظت از:

```text
Laptop
PC
Mobile
```

---

<p dir='rtl' style='font-size:24px; font-wight:bold;'>
Information Security
</p>

امنیت اطلاعات

---

محافظت از:

```text
Data
```

---

صرف‌نظر از محل ذخیره‌سازی آن.

---

# تهدید چیست؟

در امنیت اصطلاحی داریم:

```text
Threat
```

---

<p dir='rtl'>
Threat یعنی:
</p>

```text
هر چیزی که بتواند آسیب وارد کند
```

---

مثال:

```text
Hacker
Malware
Virus
Ransomware
Insider Threat
```

---

# آسیب‌پذیری چیست؟

اصطلاح:

```text
Vulnerability
```

---

به معنای:

```text
ضعف امنیتی
```

است.

---

مثال:

---

فرض کن رمز عبور:

```text
123456
```

باشد.

---

این یک:

```text
Vulnerability
```

است.

---

# حمله چیست؟

اصطلاح:

```text
Attack
```

---

زمانی رخ می‌دهد که مهاجم از یک ضعف استفاده کند.

---

مثال:

```text
Weak Password
```

↓

```text
Password Guessing
```

↓

```text
Unauthorized Access
```

---

# امنیت کامل وجود دارد؟

خیر.

---

یکی از مهم‌ترین مفاهیم امنیت:

```text
Perfect Security Does Not Exist
```

---

هیچ سیستمی 100٪ امن نیست.

---

هدف امنیت:

```text
کاهش ریسک
```

است.

---

# ریسک چیست؟

اصطلاح:

```text
Risk
```

---

به معنای:

```text
احتمال وقوع آسیب
```

است.

---

فرمول ساده:

```text
Risk = Threat × Vulnerability
```

---

هرچه ضعف بیشتر باشد:

---

ریسک بیشتر خواهد شد.

---

# نقش ما در این پروژه چیست؟

پروژه‌ای که ساخته‌ایم:

```text
Port Scanner
```

---

بخشی از فرآیند:

```text
Reconnaissance
```

است.

---

یعنی:

```text
جمع‌آوری اطلاعات
```

---

قبل از هر تحلیل امنیتی باید بدانیم:

```text
چه سرویس‌هایی فعال هستند؟
```

---

پورت اسکنر دقیقاً همین کار را انجام می‌دهد.

---

# آیا پورت اسکنر ابزار هک است؟

خیر.

---

پورت اسکنر یک ابزار است.

---

مانند:

```text
چاقوی آشپزخانه
```

---

می‌توان از آن:

```text
درست استفاده کرد
```

یا

```text
نادرست استفاده کرد
```

---

کاربرد ابزار وابسته به نحوه استفاده از آن است.
